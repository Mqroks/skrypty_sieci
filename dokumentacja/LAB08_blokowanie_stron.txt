# ============================================
# 1. KONFIGURACJA ADRESU WAN
# ============================================
/ip/dhcp-client
# ether1 - adres pobierany z DHCP sali (z DNS i NTP od dostawcy)
add interface=ether1 use-peer-dns=yes use-peer-ntp=yes add-default-route=yes

# ============================================
# 2. NAT - MASKARADA DLA DOSTĘPU DO INTERNETU
# ============================================
/ip/firewall/nat
# Translacja adresów dla ruchu wychodzącego przez ether1 (WAN)
add action=masquerade chain=srcnat out-interface=ether1

# ============================================
# 3. KONFIGURACJA ADRESU IP NA INTERFEJSIE LAN
# ============================================
/ip/address
# ether2 - sieć LAN lokalna
add interface=ether2 address=192.168.2.1/24

# ============================================
# 4. KONFIGURACJA DHCP DLA SIECI LAN
# ============================================
/ip/pool
# Pula adresów dla klientów LAN
add name=dhcp1 ranges=192.168.2.10-192.168.2.100

/ip/dhcp-server/network
# Parametry sieci DHCP (brama i serwer DNS)
add address=192.168.2.0/24 gateway=192.168.2.1 dns-server=8.8.8.8

/ip/dhcp-server
# Uruchomienie serwera DHCP na interfejsie ether2
add name=DHCP interface=ether2 address-pool=dhcp1

# ============================================
# 5. BLOKADA STRONY - LAYER7 FIREWALL
# ============================================
/ip/firewall/layer7-protocol
# Definicja wzorca URL do blokowania (ZSL Kraków)
add name=Blokada_ZSL regexp=https://tl.krakow.pl/

/ip/firewall/filter
# Blokada ruchu HTTPS do dopasowanych domen (Layer7)
add chain=forward protocol=tcp dst-port=443 layer7-protocol=Blokada_ZSL action=drop
