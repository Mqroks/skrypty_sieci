############################################
# KONFIGURACJA ROUTERA R2                  #
############################################

# ============================================
# 1. ADRESACJA IP INTERFEJSÓW
# ============================================
# Adres IP dla sieci LAN na porcie ether1
/ip/address/ add address=192.168.20.1/24 interface=ether1

# Adres IP dla połączenia z Routerem R3 na porcie ether2
/ip/address/ add address=172.16.30.2/24 interface=ether2

# Adres IP dla połączenia z Routerem R1 na porcie ether3
/ip/address/ add address=172.16.10.2/24 interface=ether3

# ============================================
# 2. KONFIGURACJA SERWERA DHCP
# ============================================
/ip/pool/ add name=dhcp_pool ranges=192.168.20.2-192.168.20.254
/ip/dhcp-server/ add name=dhcp1 interface=ether1 address-pool=dhcp_pool

# ============================================
# 3. ROUTING DYNAMICZNY RIPv2
# ============================================
/routing/rip/instance/ add name=RIPv2 redistribute=connected

# Aktywacja RIP na interfejsach lokalnych i między-routerowych:
/routing/rip/interface-template/ add instance=RIPv2 interfaces=ether1 source-addresses=192.168.20.0 cost=1
/routing/rip/interface-template/ add instance=RIPv2 interfaces=ether2 source-addresses=172.16.30.0 cost=1
/routing/rip/interface-template/ add instance=RIPv2 interfaces=ether3 source-addresses=172.16.10.0 cost=1
