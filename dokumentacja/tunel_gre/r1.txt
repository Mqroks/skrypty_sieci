################################################
# KONFIGURACJA ROUTERA R1 (ODDZIAŁ 1)          #
################################################

# 1. Pobieranie adresu publicznego z DHCP na porcie WAN (ether1).
/ip/dhcp-client/ add interface=ether1 

# 2. Konfiguracja sieci lokalnej LAN na porcie ether3.
/ip/address/ add interface=ether3 address=192.168.10.1/24 
/ip/pool/ add name=dhcp_pool ranges=192.168.10.2-192.168.10.254 
/ip/dhcp-server/ add name=dhcp1 interface=ether3 address-pool=dhcp_pool 

# 3. NAT (Maskarada) dla wyjścia na świat przez ether1.
/ip/firewall/nat/ add chain=srcnat out-interface=ether1 action=masquerade 

# 4. TUNEL GRE + IPsec:
# Tworzenie tunelu do routera R2 (70.70.70.1) zabezpieczonego hasłem "Haslo12345".
/interface/gre/ add name=gre-tunel local-address=80.80.80.1 remote-address=70.70.70.1 ipsec-secret=Haslo12345 allow-fast-path=no 

# 5. Adresacja wewnątrz tunelu (punkt-punkt).
/ip/address/ add address=10.10.10.1/24 interface=gre-tunel 

# 6. Routing statyczny: kierowanie ruchu do sieci LAN routera R2 przez tunel.
/ip/route/ add dst-address=192.168.20.0/24 gateway=10.10.10.2
