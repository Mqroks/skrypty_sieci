################################################
# KONFIGURACJA ROUTERA R2 (ODDZIAŁ 2)          #
################################################

# 1. Pobieranie adresu publicznego na porcie WAN.
/ip/dhcp-client/ add interface=ether1 

# 2. Lokalna sieć LAN na porcie ether3 (inna podsieć: 20.0/24).
/ip/address/ add interface=ether3 address=192.168.20.1/24 
/ip/pool/ add name=dhcp_pool ranges=192.168.20.2-192.168.20.254 
/ip/dhcp-server/ add name=dhcp1 interface=ether3 address-pool=dhcp_pool 

# 3. Wyjście na internet.
/ip/firewall/nat/ add chain=srcnat out-interface=ether1 action=masquerade 

# 4. TUNEL GRE + IPsec:
# Nawiązanie połączenia powrotnego do routera R1 (80.80.80.1).
/interface/gre/ add name=gre-tunel local-address=70.70.70.1 remote-address=80.80.80.1 ipsec-secret=Haslo12345 allow-fast-path=no 

# 5. Adresacja wewnątrz tunelu.
/ip/address/ add address=10.10.10.2/24 interface=gre-tunel 

# 6. Routing statyczny: kierowanie ruchu do sieci LAN routera R1 przez tunel.
/ip/route/ add dst-address=192.168.10.0/24 gateway=10.10.10.1
