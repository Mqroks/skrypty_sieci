#Dodanie bridge
interface/bridge/add name=LAN protocol-mode=none vlan-filtering=yes 


#Porty do bridge
interface/bridge/port/add bridge=LAN interface=ether1 pvid=1
interface/bridge/port/add bridge=LAN interface=ether2 pvid=10
interface/bridge/port/add bridge=LAN interface=ether3 pvid=20

#konfiguracja VLANów
interface/bridge/vlan/add bridge=LAN vlan-ids=1 tagged=ether1,LAN  untagged=none  
interface/bridge/vlan/add bridge=LAN vlan-ids=10 tagged=ether1 untagged=ether2
interface/bridge/vlan/add bridge=LAN vlan-ids=20 tagged=ether1 untagged=ether3
