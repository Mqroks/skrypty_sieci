# KONFIGURACJA ROUTERA - MikroTik RouterOS #
############################################

# ============================================
# 1. KONFIGURACJA ADRESÓW IP NA INTERFEJSACH
# ============================================
/ip/dhcp-client
# ether4 - adres publiczny pobierany od ISP
add interface=ether4

/ip/address
# ether2 - sieć LAN lokalna
add interface=ether2 address=192.168.10.1/24

# ============================================
# 2. KONFIGURACJA DHCP DLA SIECI LAN
# ============================================
/ip/pool
# Pula adresów dla klientów LAN
add name=LAN ranges=192.168.10.100-192.168.10.150

/ip/dhcp-server/network
# Parametry sieci DHCP (brama i serwer DNS)
add address=192.168.10.0/24 gateway=192.168.10.1 dns-server=8.8.8.8

/ip/dhcp-server
# Uruchomienie serwera DHCP na interfejsie ether2
add name=LAN interface=ether2 address-pool=LAN

# ============================================
# 3. NAT - MASKARADA DLA DOSTĘPU DO INTERNETU
# ============================================
/ip/firewall/nat
# Translacja adresów dla ruchu wychodzącego przez ether4 (WAN)
add chain=srcnat out-interface=ether4 action=masquerade

# ============================================
# 4. KONFIGURACJA VPN - L2TP SERVER
# ============================================
/ip/pool
# Pula adresów przydzielanych klientom VPN
add name=VPN ranges=172.16.0.10-172.16.0.100

/interface/l2tp-server/server
# Uruchomienie serwera L2TP
set enabled=yes

/ppp/profile
# Profil PPP - lokalny adres tunelu i pula dla klientów
set default-encryption local-address=172.16.0.1 remote-address=VPN

/ppp/secret
# Dane uwierzytelniające użytkownika VPN
add name=tester password=12345 profile=default-encryption

# ============================================
# 5. FIREWALL - REGUŁY DLA SERWERA L2TP/VPN
# ============================================
/ip/firewall/filter
# Zezwolenie na ruch IKE (negocjacja kluczy IPsec)
add chain=input protocol=udp dst-port=500  comment=L2TP-Server
# Zezwolenie na ruch L2TP
add chain=input protocol=udp dst-port=1701 comment=L2TP-Server
# Zezwolenie na ruch NAT-T (IPsec przez NAT)
add chain=input protocol=udp dst-port=4500 comment=L2TP-Server
