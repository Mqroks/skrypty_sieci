############################################
# KONFIGURACJA ROUTERA R2                 #
############################################

# ============================================
# 1. ADRESY IP NA INTERFEJSACH
# ============================================
/ip/address
# Przypisanie adresu IP do interfejsu ether1:
# - Sieć LAN podłączona do R2 (192.168.20.0/24)
# - R2 pełni rolę bramy domyślnej dla tej sieci
add interface=ether1 address=192.168.20.1/24

# Przypisanie adresu IP do interfejsu ether2:
# - Łącze WAN między R2 a R3 (sieć 172.16.30.0/24)
# - R2 ma adres .2, R3 ma adres .1
add interface=ether2 address=172.16.30.2/24

# Przypisanie adresu IP do interfejsu ether3:
# - Łącze WAN między R2 a R1 (sieć 172.16.10.0/24)
# - R2 ma adres .2, R1 ma adres .1
add interface=ether3 address=172.16.10.2/24

# ============================================
# 2. PULA ADRESÓW DHCP
# ============================================
/ip/pool
# Tworzenie puli adresów IP o nazwie "pool10":
# - Zakres adresów: 192.168.20.100 - 192.168.20.200
# - Adresy będą przydzielane dynamicznie klientom sieci LAN
add name=pool10 ranges=192.168.20.100-192.168.20.200

# ============================================
# 3. KONFIGURACJA SIECI DHCP
# ============================================
/ip/dhcp-server/network
# Definicja sieci obsługiwanej przez serwer DHCP:
# - address: zakres sieci (192.168.20.0/24)
# - gateway: brama domyślna wysyłana do klientów (adres R2 na ether1)
add address=192.168.20.0/24 gateway=192.168.20.1

# ============================================
# 4. SERWER DHCP
# ============================================
/ip/dhcp-server
# Uruchomienie serwera DHCP o nazwie "dhcp10":
# - interface=ether1: nasłuchiwanie na interfejsie LAN
# - address-pool=pool10: korzystanie z wcześniej zdefiniowanej puli adresów
add name=dhcp10 interface=ether1 address-pool=pool10

# ============================================
# 5. TRASY STATYCZNE (ROUTING)
# ============================================
/ip/route/
# Trasa do sieci LAN routera R1 (192.168.10.0/24):
# - Pakiety kierowane przez łącze do R1 (brama: 172.16.10.1)
add dst-address=192.168.10.0/24 gateway=172.16.10.1

# Trasa do łącza WAN między R1 a R3 (172.16.20.0/24):
# - Dostęp do tej sieci też przez R1 (brama: 172.16.10.1)
add dst-address=172.16.20.0/24 gateway=172.16.10.1

# Trasa do sieci LAN routera R3 (192.168.30.0/24):
# - Pakiety kierowane przez łącze do R3 (brama: 172.16.30.1)
add dst-address=192.168.30.0/24 gateway=172.16.30.1

############################################
# KONIEC KONFIGURACJI ROUTERA R2          #
############################################
